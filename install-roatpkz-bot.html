STEP 1 — Install (one time)

Open PowerShell as Administrator and paste this entire block:

# Roat Pkz PVP Bot v2.0 Installer
$ErrorActionPreference = "Stop"
$dir = "$env:USERPROFILE\RoatPkzBot"
Write-Host "`n=== ROAT PKZ PVP BOT v2.0 INSTALLER ===" -ForegroundColor Red
Write-Host "Installing to: $dir`n" -ForegroundColor Gray

# Create directory
if (!(Test-Path $dir)) { New-Item -ItemType Directory -Path $dir -Force | Out-Null }

# Check Python
$py = $null
foreach ($p in @("python", "python3", "py")) {
    try { $v = & $p --version 2>&1; if ($v -match "3\.\d+") { $py = $p; break } } catch {}
}
if (!$py) {
    Write-Host "[!] Python 3 not found. Installing..." -ForegroundColor Yellow
    winget install Python.Python.3.12 --accept-package-agreements --accept-source-agreements
    $env:PATH += ";$env:LOCALAPPDATA\Programs\Python\Python312;$env:LOCALAPPDATA\Programs\Python\Python312\Scripts"
    $py = "python"
}
Write-Host "[OK] Python: $py" -ForegroundColor Green

# Install dependencies
Write-Host "`n[2/3] Installing Python packages..." -ForegroundColor Cyan
& $py -m pip install --upgrade pip --quiet
& $py -m pip install opencv-python numpy mss pynput pygetwindow Pillow --quiet
Write-Host "[OK] All packages installed" -ForegroundColor Green

# Download bot script
Write-Host "`n[3/3] Downloading bot script..." -ForegroundColor Cyan
$url = "https://raw.githubusercontent.com/rotstrucking1-prog/Rotstrucking-app1/main/roatpkz_pvp_bot_v2.py"
Invoke-WebRequest -Uri $url -OutFile "$dir\roatpkz_pvp_bot_v2.py" -UseBasicParsing
Write-Host "[OK] Bot downloaded to $dir" -ForegroundColor Green

# Create desktop shortcut
$shell = New-Object -ComObject WScript.Shell
$shortcut = $shell.CreateShortcut("$env:USERPROFILE\Desktop\Roat Pkz Bot.lnk")
$shortcut.TargetPath = (Get-Command $py).Source
$shortcut.Arguments = "`"$dir\roatpkz_pvp_bot_v2.py`""
$shortcut.WorkingDirectory = $dir
$shortcut.Description = "Roat Pkz PVP Bot v2.0"
$shortcut.Save()
Write-Host "[OK] Desktop shortcut created" -ForegroundColor Green

# Create run.bat
"@echo off`n$py `"$dir\roatpkz_pvp_bot_v2.py`"`npause" | Out-File -FilePath "$dir\run.bat" -Encoding ASCII
Write-Host "[OK] run.bat created" -ForegroundColor Green

Write-Host "`n========================================" -ForegroundColor Red
Write-Host "  INSTALLATION COMPLETE!" -ForegroundColor Green
Write-Host "  Double-click 'Roat Pkz Bot' on desktop" -ForegroundColor White
Write-Host "  Or run: $dir\run.bat" -ForegroundColor Gray
Write-Host "========================================`n" -ForegroundColor Red

1. Make sure Roat Pkz is open and you're logged in
