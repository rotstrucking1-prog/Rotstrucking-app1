📋 Setup

  1. Create a NEW package on DeadZone (or use existing empty one)
  2. Copy each tab below and paste into the matching DZ tab
  3. Stand next to the eternal fire in Rogues' Den
  4. Have raw karambwans in your inventory
  5. Enable the plugin, flip "Start Cooking" ON in the CONFIG sidebar

⚡ MAIN TAB
var RAW = 3142;
var FIRE = 43475;
var on = false;
var cnt = 0;
var lt = 0;
var tc = 0;

function botMsg(m) { Game.sendGameMessage("[K] " + m); }

function OnStart() { botMsg("Karambwan cooker loaded"); }

function OnGameTick() {
    tc++;
    var go = false;
    try { go = ("" + config.enabled.getValue()) === "true"; } catch(e) {}
    if (!go) {
        if (on) { botMsg("OFF - cooked " + cnt); on = false; }
        return;
    }
    if (!on) { on = true; cnt = 0; botMsg("ON"); }
    var has = false;
    try { has = Game.info.inventory.hasItem(RAW, 1); } catch(e) {
        try { has = Game.info.inventory.hasItem(RAW); } catch(e2) {}
    }
    if (!has) {
        botMsg("Done! " + cnt + " cooked");
        try { config.enabled.setValue(false); } catch(e) {}
        on = false;
        return;
    }
    if (tc - lt < 1) return;
    var f = null;
    try {
        var obs = Game.info.object.getAll();
        for (var i = 0; i < obs.length; i++) {
            if (obs[i].getId() === FIRE) { f = obs[i]; break; }
        }
    } catch(e) { botMsg("ERR obj: " + e); return; }
    if (f === null) { botMsg("No fire found"); return; }
    try {
        Game.interact.inventory.useItem(RAW);
        Game.interact.object.interact(f, MenuAction.GAME_OBJECT_FIRST_OPTION);
        lt = tc;
        cnt++;
        if (cnt % 28 === 0) botMsg("Cooked " + cnt);
    } catch(e) { botMsg("ERR cook: " + e); }
}
