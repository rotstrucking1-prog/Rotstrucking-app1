function botMsg(msg) {

function cfgVal(name, def) {
    try {
        var raw = config[name].getValue();
        if (raw === null || raw === undefined) return def;
        var s = "" + raw;
        if (s === "false") return "false";
        if (s === "true") return "true";
        return s;
    } catch(e) {
        return def;
    }
}

function OnStart() {
    botMsg("One Tick Karambwan loaded! Flip Start Cooking ON.");
}
